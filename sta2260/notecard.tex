		P(E) &= \text{\# $E$ occs.} \div \text{\# $S$ occs.} \\
			&\note{given all outcomes in S are equally likely} \\
		P(A \cup B) &= P(A) + P(B) - P(A \cap B) \\
		\text{Disjoint}&: P(A \cap B) = 0
			\note{mutually exclusive} \\
		\text{Indpndnt}&: P(A \cap B) = P(A) P(B) \\
		P(A|B) &= \frac{P(A \cap B)}{P(B)} \\
		P(A \cap B) &= P(A|B)P(B) \lor P(B|A)P(A) \\
		P(A) &= P(A|B)P(B) + P(A|\bar B)P(\bar B) \\
		P(B|A) &= \frac{P(B \cap A)}{P(A)} = \frac{P(A|B)P(B)}{P(A)}
	\end{align*}

	\begin{align*}
		E(X) &= \sum_{i \in R(X)} i P(X=i) = \mu \\
		Var(X) &= E(X^2)-\mu^2=\sigma^2 \\
		&{\small{E(X)^2 = \sum_{i \in R(X)} i^2 P(X=i)}} \\
		\text{B:} &
		\begin{cases}
			R(x) &= \{ 0, 1, ..., n \} \\
			P(X=j) &= \binom{n}{j} p^j (1 - p)^{n-j} \\
			&\note{: Prob of $j$ succ. outof $n$ trials} \\
			E(X) &= np = \mu \\
			Var(X) &= \sqrt{np(1-p)} = \sigma
		\end{cases} \\
		\text{Poi:} &
		\begin{cases}
			R(x) &= \{ 0, 1, 2, 3,4 , ... \} \\
			P(X=j) &= \frac{e^{-\lambda}\lambda^j}{j!} \\
			&\note{: Prob of $j$ occ. in interv.} \\
			E(X) &= \lambda \\
			Var(X) &= \lambda \qquad (SD = \sqrt{\lambda})
		\end{cases} \\
		\text{G:} &
		\begin{cases}
			R(x) &= \{ 1, 2, 3, 4 , ... \} \\
			P(X=j) &= (1-p)^{j-1} p, \note{0<p<1} \\
			E(X) &= 1/p \\
			Var(X) &= (1-p)/p^2
		\end{cases}
	\end{align*}
\end{minipage}

	\begin{align*}
		\text{HG:} &
		\begin{cases}
			R(x) &= \note{max(0,n+m-N),...,min(m,n)} \\
			P(X=j) &= \frac{\binom{m}{j} \binom{N-m}{n-j}}{\binom{N}{n}} \\
			E(X) &= \frac{nm}{N} \\
			Var(X) &= \frac{nm(N-n)(N-m)}{N^2(N-1)}
		\end{cases} \\
		\text{Co:} &
		\begin{cases}
			f(x) &\ge 0 \forall x \in R(X) \\
			\int_{R(X)}^a f(x) dx &= 1 \\
			P\{ a < X < b \} &= \int_a^b f(x) dx \\
			E(X) &= \int_{R(X)} x f(x) dx = \mu \\
			Var(X) &= E(X^2) - \mu^2 \\
			f(X &= a) \ne \textit{prob!}
		\end{cases} \\
		\text{Exp:} &
		\begin{cases}
			f(x) &= \lambda e^{- \lambda x} \text{ where } x \ge 0 \\
			E(x) &= \frac{1}{x} \\
			Var(x) &= \frac{1}{x^2} \\
		\end{cases} \\
		\text{Norm:} &
		\begin{cases}
			f(x) &= \frac{1}{ \sigma \sqrt{2 \pi} } e^{ - \frac{1}{2\sigma^2}(X-\mu)^2} \\
			\text{Mean} &= \mu \text{ (location)} \text{, SD} = \sigma \text{ (shape)} \\
		\end{cases} \\
		Z_\alpha &= 100(1-\alpha)^\text{th} \text{prcnt} \\
		& (X - \mu) \div \sigma = Z \sim N(0,1) \\
		& \sigma Z + \mu = X \sim N(\mu,\sigma)
	\end{align*}
