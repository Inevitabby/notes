\begin{minipage}{0.48\textwidth}
	\begin{align*}
		\bar{X} &= \hat{\mu}
			, s = \hat{\sigma} 
			, SE_{\bar{X}} = s \div \sqrt{n} \\
		n &= \left( z_{\alpha / 2} \sigma \div SE \right)^2 \\
		\text{C.I. of $\mu$: }& \bar{X} \pm t_{\alpha / 2, n - 1} SE_{\bar{X}} \\
		\textbf{one} \\
		\tiny\textit{1 set $\mu, H, \alpha$; }
		&\tiny\textit{2 check CLT / norm; }
		\tiny\textit{3 calc $\bar{X}$, $s$; then $T$}
		\\
		T &= \frac{
			\bar{X} - c
		}{
			s / \sqrt{n}
		} \sim t(n-1) \\
		\text{Reject $H_0$ if:}&
		\begin{cases}
			H_1 : u \ne c & |T| > t_{\alpha / 2 , n - 1} \\
				& p = 2 P(t > |T|) \\
			H_1 : u > c & T > t_{\alpha , n - 1} \\
				& p = P(t > T) \\
			H_1 : u < c & T < - t_{\alpha , n - 1} \\
				& p = P(t < T)
		\end{cases} \\
		&\qquad \tiny\textit{also rej. $H_0$ if p-value < $\alpha$} \\
		\textbf{forgetful} \\
		s^2 &= \frac{ \sum_{i=1}^n x_i^2 - n \bar{x}^2 }{ n - 1 }
	\end{align*}
\end{minipage}
\begin{minipage}{0.48\textwidth}
	\begin{align*}
		\textbf{2 independent} \\
		\tiny\textit{$\sigma_1 = \sigma_2 \text{ if } \frac{ max(s_1, s_2) }{ min(s_1, s_2) } < 2$} \\
		\begin{aligned}
			\sigma_1 = \sigma_2 &:
			\begin{cases}
				T &= \frac{\bar{X}_1 - \bar{X}_2}{SE_{\text{pooled}}}
				\\
				df^* &= n_1 + n_2 - 2
			\end{cases}
			\\
			\sigma_1 \ne \sigma_2 &:
			\begin{cases}
				T &= \frac{\bar{X}_1 - \bar{X}_2}{SE_{\text{unpooled}}}
				\\
				df^* &= min(n_1 - 1, n_2 - 1)
			\end{cases}
		\end{aligned} \\
		\tiny\textit{$SE_{\text{u..}} = \sqrt{\frac{s_1^2}{n_1} + \frac{s_2^2}{n_2}}$, }
		\tiny\textit{$SE_{\text{p..}} = \sqrt{S^2_{\text{p..}}\left(\frac{1}{n_1} + \frac{1}{n_2}\right)}$}\\
		\tiny\textit{$S^2_{\text{p..}} = \frac{(n_1 - 1)s_1^2 + (n_2 - 1)s_2^2}{n_1 + n_2 - 2} \quad$}\\
		\text{Reject $H_0$ if:}
		\begin{cases}
			H_1 : \mu_1 - \mu_2 \ne 0 & |T| > t_{\alpha / 2 , df}
			\\
			H_1 : \mu_1 - \mu_2 > 0 & |T| > t_{\alpha , df}
			\\
			H_1 : \mu_1 - \mu_2 < 0 & |T| < - t_{\alpha , df}
		\end{cases} \\
		\text{indp C.I.: } (\bar{X}_1 - \bar{X}_2) \pm t_{\alpha / 2 , df} SE_? \\
		\tiny\textit{accept $H_0$ if has zero} \\
	\end{align*}
\end{minipage}

	\begin{align*}
		\textbf{2 paired} \\
		\text{Paired: }
		\begin{cases}
			H_0 : \text{use } \mu_d \\
			d_i = x_{1i} - x_{2i} \forall i=1,...,n \\
			\text{compute } \bar{d}, s_d, SE_{\bar{d}} \\
			SE_{\bar{d}} = s_d / \sqrt{n} \\
			T = \frac{\bar{d}}{SE_{\bar{d}}} \sim t(n-1)
		\end{cases} \\
		\text{Reject $H_0$ if: }
		\begin{cases}
			H_1 : \mu_d \ne 0 & |T| > t_{\alpha / 2 , n - 1}
			\\
			H_1 : \mu_d > 0 & T > t_{\alpha , n - 1}
			\\
			H_1 : \mu_d < 0 & T < - t_{\alpha , n - 1}
		\end{cases} \\
		\text{pair C.I.: } \bar{d} \pm t_{\alpha / 2, n-1} SE_{\bar{d}} \\
		\tiny\textit{accept $H_0$ if has zero} \\
		\textbf{pop param} \\
		\text{$\hat p$: }
		\begin{cases}
			\hat p &= \frac{X}{n} \\
			\hat p &\sim N( p, \sqrt{ \frac{p(1-p)}{n} } )
		\end{cases} \\
		\tiny\textit{$X$: \# of outcms w/ a char.} \\
		\tiny\textit{1 set $p, H, \alpha$; }
		\tiny\textit{2 check $np$ and $n(1-p) \ge 5$; }
		\tiny\textit{3 calc $\hat p$; then $Z$} \\
		Z = \frac{ \hat p - p_0 }{ \sqrt{ \frac{ p_0 (1-p_0) }{ n } } } \sim N(0,1) \\
		\tiny\textit{$p = p_0$ under $H_0$} 
	\end{align*}
